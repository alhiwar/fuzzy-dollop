الأدب

ضفتان ولا جسر

يحيى السماوي

يحيى السماوي

الأدب

تاريخ النشر: 2026-10-10

في آخر تكبيرةِ فجرٍ

من شعبانْ

في عامِ الفيلِ القوميِّ

طَوَيْتُ بساطي..

قلتُ لأمي:

حفنة أيامٍ

وتعودُ حمامةُ قلبي لروابيكِ..

تعودُ الخضرةُ للأَفْنانْ

رَشَّتْ حولي ماءً عَفِناً

(كنّا نشربُ من تُرْعَةِ بستانْ)

وأنا كنتُ رَشَشْتُ يديها بالقُبُلاتِ

وحين هَمَمْتُ بلثم صغيري

صَرَخَتْ زوجي:

دَخَلَ الجندُ الحيَّ!

فَلَمْلَمْتُ بقايايَ..

وكالنَسْناسِ

قفزتُ إلى سطحِ الجيرانْ

لم يمهلْني الرعبُ طويلاً

فَتَعَكَّزْتُ ضلوعي قبل طلوعِ الشمسِ

طريداً

أبحثُ عن واحاتِ أمانْ

مَرَّ الليلُ الأولُ في قريةِ «سيِّد جبار»...

الثاني؟

في الصحراءِ..

الثالثُ؟

في «سفوانْ»

ثُمَّ توارى النخلُ... المدن الثكلى..

والخِلاّنْ

***

أقفُ الآنْ

في منعطف العمرِ غريباً

بين صديقٍ لا أعرفُهُ..

وعدوٍّ يعرفني..

حيناً تحرقني الأمطارُ

وحيناً

تُثْلِجُني النيرانْ..

أقف الآنْ

بين صديقٍ لا يعرفني

وعدوٍّ أعرفُهُ..

بينَ عبيرِ زهورِ الروحِ

وأشواكِ الجسدِ الشيطانْ

مَنْ ينقذني مني؟

فَيُوَحِّدُ ما بين يقينِ جذوري

وظنونِ الأغصانْ؟

فأنا الانْ

ضِفَتانِ ولا جسرٌ..

نصفي وحشٌ -

والآخرُ انسانْ!

***

في آخرِ تكبيرة فجرٍ من شعبانْ

من عامِ الفيلِ القوميِّ

ابتدأَ الرحلةَ رجلٌ في مُقْتَبَلِ العشقِ..

وزوجٌ .. وصغيرانْ
